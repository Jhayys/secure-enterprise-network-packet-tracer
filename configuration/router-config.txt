! ============================================================
! JHAYYS DIGITAL SERVICES
! Secure Small-Business Network Lab
! CORE-ROUTER Configuration
! Cisco 2911 - Packet Tracer
! ============================================================

version 15.1

hostname CORE-ROUTER

! ————————————————————
! DHCP CONFIGURATION
! ————————————————————

ip dhcp excluded-address 192.168.10.1 192.168.10.20
ip dhcp excluded-address 192.168.40.1 192.168.40.20

ip dhcp pool STAFF
network 192.168.10.0 255.255.255.0
default-router 192.168.10.1
dns-server 8.8.8.8

ip dhcp pool GUEST
network 192.168.20.0 255.255.255.0
default-router 192.168.20.1
dns-server 8.8.8.8

ip dhcp pool IT-ADMIN
network 192.168.30.0 255.255.255.0
default-router 192.168.30.1
dns-server 8.8.8.8

! ————————————————————
! DOMAIN / USER CONFIGURATION
! ————————————————————

ip domain-name jhayys.local

! Local administrative credential intentionally redacted.
! username netadmin privilege 15 secret 

! ————————————————————
! ROUTER-ON-A-STICK / VLAN GATEWAYS
! ————————————————————

interface GigabitEthernet0/0
no ip address
duplex auto
speed auto

interface GigabitEthernet0/0.10
encapsulation dot1Q 10
ip address 192.168.10.1 255.255.255.0
ip access-group STAFF-RESTRICTIONS in

interface GigabitEthernet0/0.20
encapsulation dot1Q 20
ip address 192.168.20.1 255.255.255.0
ip access-group GUEST-RESTRICTIONS in

interface GigabitEthernet0/0.30
encapsulation dot1Q 30
ip address 192.168.30.1 255.255.255.0

interface GigabitEthernet0/0.40
encapsulation dot1Q 40
ip address 192.168.40.1 255.255.255.0

! ————————————————————
! UNUSED INTERFACES
! ————————————————————

interface GigabitEthernet0/1
no ip address
duplex auto
speed auto
shutdown

interface GigabitEthernet0/2
no ip address
duplex auto
speed auto
shutdown

interface Vlan1
no ip address
shutdown

! ————————————————————
! ACCESS CONTROL LISTS
! ————————————————————

ip access-list extended GUEST-RESTRICTIONS
deny ip 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255
deny ip 192.168.20.0 0.0.0.255 192.168.40.0 0.0.0.255
permit ip 192.168.20.0 0.0.0.255 any

ip access-list extended STAFF-RESTRICTIONS
deny ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
permit ip 192.168.10.0 0.0.0.255 any

ip access-list standard MGMT-ACCESS
permit 192.168.30.0 0.0.0.255

! ————————————————————
! SECURE REMOTE MANAGEMENT
! ————————————————————

line vty 0 4
access-class MGMT-ACCESS in
login local
transport input ssh

! ============================================================
! END OF CORE-ROUTER CONFIGURATION
! ============================================================
